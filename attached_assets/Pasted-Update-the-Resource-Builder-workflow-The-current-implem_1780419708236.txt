## HELPER FUNCTION

function getNextCategory(category) {
const categories = [
"Academic Paper",
"Book",
"Government Report",
"Research Study",
"Competitor Book",
"Case Study",
"Statistics Source",
"Interview / Transcript",
"Blog / Article",
"Writing Style Reference",
"Note / Finding",
"Other"
];

const index = categories.indexOf(category);

if (index === -1) return category;
if (index === categories.length - 1) return category;

return categories[index + 1];
}
